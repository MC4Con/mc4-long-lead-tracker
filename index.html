const hiddenProjects = PropertiesService.getScriptProperties().getProperty('ll_hidden_projects') || ''; return { items, categories, hiddenProjects }; } function saveAll(sheet, items, categories, hiddenProjects) { if (categories) { PropertiesService.getScriptProperties().setProperty('ll_categories', categories);
